%% Low-rank adaptation in one picture: the pretrained weight matrix stays frozen;
%% a low-rank additive update B*A is trained beside it; both paths sum at the output.
%%
%% Hand-authored TikZ (not generated): it depicts the published method (Hu et al.
%% 2021), cited in the accompanying prose, not a project measurement.
%%
%% Input from a chapter with:  \input{figures/lora-update.tex}

\begin{tikzpicture}[
    font=\small,
    frozen/.style={
      draw=horusaccent, line width=0.8pt, rounded corners=2pt,
      fill=horuswash, align=center, inner sep=0pt,
    },
    trained/.style={
      draw=horuswarn, line width=0.8pt, rounded corners=2pt,
      fill=white, align=center, inner sep=0pt,
    },
    flow/.style={-{Stealth[length=2.2mm]}, line width=0.7pt, draw=horusink},
    lbl/.style={font=\scriptsize, text=horusink, align=center},
  ]

  \coordinate (in) at (0, 0);
  \node[lbl, anchor=east] at (in) {input $x$};

  %% The frozen full-rank matrix: large square, wash fill.
  \node[frozen, minimum width=2.2cm, minimum height=2.2cm] (W) at (2.6, 1.0)
    {$W$\\[1pt]{\scriptsize frozen}};

  %% The trained low-rank pair: two thin rectangles, warn border. Their thinness IS
  %% the message -- the inner dimension r is far smaller than the matrix dimensions.
  \node[trained, minimum width=2.2cm, minimum height=0.5cm] (A) at (2.6, -1.15)
    {\scriptsize $A$ \ (rank $r$)};
  \node[trained, minimum width=0.5cm, minimum height=1.1cm] (B) at (4.35, -1.15)
    {\scriptsize $B$};

  \coordinate (sum) at (6.0, 0);
  \node[draw=horusink, circle, inner sep=1pt, line width=0.7pt] (plus) at (sum) {$+$};

  \draw[flow] (in) -- (1.0, 0) |- (W.west);
  \draw[flow] (in) -- (1.0, 0) |- (A.west);
  \draw[flow] (W.east) -| (plus.north);
  \draw[flow] (A.east) -- (B.west);
  \draw[flow] (B.east) -| (plus.south);
  \draw[flow] (plus.east) -- ++(1.3, 0) node[lbl, anchor=west]
    {output $Wx + BAx$};

  \node[lbl, anchor=north, text width=5.4cm] at (3.5, -2.1)
    {only $A$ and $B$ are trained and stored;\\the update $\Delta W = BA$ has rank
    at most $r$};

\end{tikzpicture}
